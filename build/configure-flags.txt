# One ScummVM ./configure flag per line. Lines starting with # are ignored.
# The workflow adds --host=android-arm-v7a --disable-debug --enable-release itself.

# Only the SCUMM engine: v0-v6 games (Monkey Island 1-2, Indy 3-4, Loom,
# Day of the Tentacle, Sam & Max, ...) plus v7/v8 (Full Throttle, The Dig,
# Curse of Monkey Island). Sub-engines are NOT enabled by --enable-engine=scumm
# when --disable-all-engines is used, so v7/v8 needs its own flag (hyphens).
--disable-all-engines
--enable-engine=scumm
--enable-engine=scumm-7-8
# Leave out Humongous Entertainment games and Rebel Assault 2 PSX.
--disable-engine=he
--disable-engine=rebel2_psx

# Features we do not need.
--disable-fluidsynth
--disable-mt32emu
--disable-tts
--disable-cloud
--disable-libcurl
--disable-sdlnet
--disable-translation
